SELECT
    DB_NAME(mf.database_id) AS DatabaseName,
    mf.name AS LogicalFileName,
    mf.physical_name AS PhysicalFileName,

    CAST(mf.size / 128.0 AS DECIMAL(18,2))
        AS CurrentSizeMB,

    CAST(ls.total_log_size_mb AS DECIMAL(18,2))
        AS TotalLogSizeMB,

    CAST(ls.active_log_size_mb AS DECIMAL(18,2))
        AS ActiveLogSizeMB,

    CAST(
        ls.total_log_size_mb - ls.active_log_size_mb
        AS DECIMAL(18,2)
    ) AS InactiveLogSizeMB,

    CAST(
        100.0 * ls.active_log_size_mb /
        NULLIF(ls.total_log_size_mb, 0)
        AS DECIMAL(10,2)
    ) AS ActiveLogPct,

    ls.total_vlf_count AS TotalVLFCount,
    ls.active_vlf_count AS ActiveVLFCount,

    ls.log_truncation_holdup_reason AS LogReuseWait

FROM sys.master_files AS mf
CROSS APPLY sys.dm_db_log_stats(mf.database_id) AS ls

WHERE mf.type_desc = 'LOG'
  AND DB_NAME(mf.database_id) IN
      ('msdb', 'ReportServerTempDB')

ORDER BY DatabaseName;

SELECT
    DB_NAME(mf.database_id) AS DatabaseName,
    mf.name AS LogicalFileName,

    COUNT(*) AS TotalVLF,

    SUM(CASE WHEN vlf.vlf_active = 1
             THEN 1 ELSE 0 END) AS ActiveVLF,

    CAST(MAX(
        CASE WHEN vlf.vlf_active = 1
             THEN vlf.vlf_begin_offset +
                  vlf.vlf_size_mb * 1048576.0
        END) / 1048576.0 AS DECIMAL(18,2))
        AS LastActiveVLFEndMB,

    CAST(mf.size / 128.0 AS DECIMAL(18,2))
        AS CurrentFileSizeMB,

    CAST(
        mf.size / 128.0 -
        MAX(CASE WHEN vlf.vlf_active = 1
                 THEN vlf.vlf_begin_offset / 1048576.0
                      + vlf.vlf_size_mb
            END)
        AS DECIMAL(18,2))
        AS PotentialShrinkMB

FROM sys.master_files AS mf
CROSS APPLY sys.dm_db_log_info(mf.database_id) AS vlf

WHERE mf.type_desc = 'LOG'
  AND DB_NAME(mf.database_id) IN
      ('msdb', 'ReportServerTempDB')

GROUP BY
    mf.database_id,
    mf.name,
    mf.size

ORDER BY DatabaseName;

SELECT
    DB_NAME(database_id) AS DatabaseName,
    name AS LogicalFileName,
    CAST(size / 128.0 AS DECIMAL(18,2)) AS CurrentSizeMB,
    CASE
        WHEN is_percent_growth = 1
            THEN CONCAT(growth, '%')
        ELSE CONCAT(growth / 128, ' MB')
    END AS AutoGrowth,
    CASE
        WHEN max_size = -1 THEN 'UNLIMITED'
        ELSE CONCAT(max_size / 128, ' MB')
    END AS MaxSize
FROM sys.master_files
WHERE type_desc = 'LOG'
  AND DB_NAME(database_id) IN
      ('msdb', 'ReportServerTempDB');
