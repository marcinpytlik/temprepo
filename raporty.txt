USE msdb;
GO

;WITH LastJobRun AS
(
    SELECT
        job_id,
        instance_id,
        run_date,
        run_time,
        run_status,
        ROW_NUMBER() OVER
        (
            PARTITION BY job_id
            ORDER BY instance_id DESC
        ) AS rn
    FROM dbo.sysjobhistory
    WHERE step_id = 0
)
SELECT
    j.job_id AS JobID,
    j.name AS JobName,
    j.enabled AS IsEnabled,

    c.Name AS ReportName,
    c.Path AS ReportPath,

    s.Description AS SubscriptionDescription,
    s.LastStatus AS SubscriptionLastStatus,
    s.LastRunTime AS SubscriptionLastRunTime,

    CASE
        WHEN h.run_date > 0
        THEN msdb.dbo.agent_datetime(h.run_date, h.run_time)
    END AS JobLastRunDate,

    h.run_status AS JobLastRunStatus

FROM dbo.sysjobs AS j

LEFT JOIN LastJobRun AS h
    ON h.job_id = j.job_id
   AND h.rn = 1

LEFT JOIN ReportServer.dbo.ReportSchedule AS rs
    ON j.name = CONVERT(NVARCHAR(128), rs.ScheduleID)

LEFT JOIN ReportServer.dbo.Subscriptions AS s
    ON rs.SubscriptionID = s.SubscriptionID

LEFT JOIN ReportServer.dbo.Catalog AS c
    ON s.Report_OID = c.ItemID

WHERE h.run_status IS NULL

ORDER BY
    c.Path,
    j.name;
